// ================================================
// PARAMETERS (Resin / SLA Optimized)
// ================================================

$fn = 120; // High resolution suited for SLA printers

// Base Parameters
base_diameter  = 25;   // Base cylinder diameter (2.5 cm)
base_height    = 15;   // Total height of the base
base_wall      = 2;    // Wall thickness for hollow base (0 for solid)

// Outer Ring Parameters
ring_wall_add  = 3;    // Ring is 3 mm wider than base
ring_height    = 12;   // Height of outer ring
knurl_count    = 24;   // Number of outer grip ridges
knurl_depth    = 0.6;  // Depth of grip ridges (mm)

// Lock & SLA Tolerances
clearance      = 0.12; // Tight gap ideal for resin printing (0.10 - 0.15mm)
pin_radius     = 1.1;  // Pin radius (2.2 mm pin diameter)
channel_depth  = 1.0;  // Channel cut depth on base

// Derived Variables
base_radius       = base_diameter / 2;
ring_inner_radius = base_radius + clearance;
ring_outer_radius = ring_inner_radius + ring_wall_add;

// ================================================
// DISPLAY / LAYOUT
// ================================================

translate([-base_radius - 6, 0, 0]) base_with_channels();
translate([ring_outer_radius + 6, 0, 0]) ring_with_pins();

// ================================================
// MODULES
// ================================================

module base_with_channels() {
    difference() {
        // Main base cylinder with top chamfer (helps guide the ring)
        chamfered_cylinder(h = base_height, r = base_radius, chamfer = 0.8);
        
        // Internal cavity (if hollowed)
        if (base_wall > 0) {
            translate([0, 0, base_wall])
                cylinder(h = base_height, r = base_radius - base_wall);
        }
        
        // L-shaped bayonet channels (180 deg apart)
        for (a = [0, 180]) {
            rotate([0, 0, a]) {
                // Vertical entry slot from top down
                translate([base_radius - channel_depth, 0, base_height - 7])
                    cylinder(h = 8, r = pin_radius + clearance);
                
                // Horizontal locking slot
                rotate_extrude(angle = 35) {
                    translate([base_radius - channel_depth, base_height - 7])
                        circle(r = pin_radius + clearance);
                }
                
                // Micro detent bump at channel end
                rotate([0, 0, 32])
                    translate([base_radius - channel_depth + 0.25, 0, base_height - 7])
                        sphere(r = pin_radius * 0.65);
            }
        }
    }
}

module ring_with_pins() {
    union() {
        // Hollow knurled ring body
        difference() {
            union() {
                // Outer ring body
                cylinder(h = ring_height, r = ring_outer_radius);
                
                // Grip ridges (knurling) around outer perimeter
                for (i = [0 : knurl_count - 1]) {
                    rotate([0, 0, i * (360 / knurl_count)])
                        translate([ring_outer_radius, 0, 0])
                            cylinder(h = ring_height, r = knurl_depth, $fn = 16);
                }
            }
            
            // Inner bore hole
            translate([0, 0, -1])
                cylinder(h = ring_height + 2, r = ring_inner_radius);
                
            // Inner chamfer at bottom opening for smooth alignment
            translate([0, 0, -0.1])
                cylinder(h = 1.0, r1 = ring_inner_radius + 0.8, r2 = ring_inner_radius);
        }
        
        // Two smooth hemispherical/rounded pins inside the ring
        for (a = [0, 180]) {
            rotate([0, 0, a]) {
                translate([ring_inner_radius, 0, ring_height - 5])
                    rotate([0, -90, 0])
                        sphere(r = pin_radius);
            }
        }
    }
}

// Helper module for chamfered cylinders (avoids sharp SLA edges)
module chamfered_cylinder(h, r, chamfer) {
    rotate_extrude() {
        polygon(points = [
            [0, 0],
            [r, 0],
            [r, h - chamfer],
            [r - chamfer, h],
            [0, h]
        ]);
    }
}
