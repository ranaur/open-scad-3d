// ================================================
// PARAMETERS (Resin SLA - Square Thread + Stopper)
// ================================================

$fn = 120; // High circular resolution

// Base Parameters
base_diameter  = 25;   // Base thread cylinder diameter (2.5 cm)
base_height    = 18;   // Base total height
thread_height  = 10;   // Height of the threaded top section
base_wall      = 2;    // Wall thickness for hollow core (0 for solid)

// Stopper Ridge Parameters
stopper_dia    = 33;   // Stopper ridge diameter (larger than ring's outer diameter)
stopper_height = 3;    // Height/thickness of the bottom stopper flange

// Square Thread Profile Parameters
thread_pitch   = 3.0;  // Vertical distance per full turn (3mm pitch)
thread_depth   = 1.0;  // Square tooth protrusion/depth (mm)

// Outer Ring Parameters
ring_wall_add  = 3;    // Ring wall thickness addition (3mm per side)
ring_height    = 10;   // Height of the threaded outer ring
knurl_count    = 24;   // Outer grip ridges count
knurl_depth    = 0.6;  // Outer grip ridges depth (mm)

// SLA Print Tolerance
clearance      = 0.15; // Radial clearance gap (0.12 - 0.15mm ideal for resin threads)

// Derived Variables
base_radius       = base_diameter / 2;
ring_inner_radius = base_radius + clearance;
ring_outer_radius = ring_inner_radius + ring_wall_add;

// ================================================
// DISPLAY LAYOUT
// ================================================

translate([-stopper_dia/2 - 5, 0, 0]) base_with_square_thread();
translate([stopper_dia/2 + 5, 0, 0]) ring_with_female_square_thread();

// ================================================
// MODULES
// ================================================

// Base cylinder with bottom stopper ridge and male square thread
module base_with_square_thread() {
    difference() {
        union() {
            // 1. Bottom Stopper Ridge / Flange
            cylinder(h = stopper_height, r = stopper_dia / 2);
            
            // 2. Main Lower Body (smooth neck above stopper)
            translate([0, 0, stopper_height])
                cylinder(h = base_height - thread_height - stopper_height, r = base_radius);
            
            // 3. Threaded Upper Section (Male Square Thread)
            translate([0, 0, base_height - thread_height])
                square_thread(
                    r = base_radius, 
                    h = thread_height, 
                    pitch = thread_pitch, 
                    depth = thread_depth,
                    internal = false, 
                    clearance = 0
                );
        }
        
        // Internal hollow cavity (if enabled)
        if (base_wall > 0) {
            translate([0, 0, base_wall])
                cylinder(h = base_height + 1, r = base_radius - base_wall);
        }
    }
}

// Outer ring with matching female square thread
module ring_with_female_square_thread() {
    difference() {
        union() {
            // Outer ring shell body
            cylinder(h = ring_height, r = ring_outer_radius);
            
            // Exterior grip ridges (knurling)
            for (i = [0 : knurl_count - 1]) {
                rotate([0, 0, i * (360 / knurl_count)])
                    translate([ring_outer_radius, 0, 0])
                        cylinder(h = ring_height, r = knurl_depth, $fn = 16);
            }
        }
        
        // Internal Female Square Thread Bore (with radial clearance applied)
        translate([0, 0, -0.1])
            square_thread(
                r = ring_inner_radius, 
                h = ring_height + 0.2, 
                pitch = thread_pitch, 
                depth = thread_depth,
                internal = true, 
                clearance = clearance
            );
            
        // Lead-in chamfer at the bottom entry for smooth threading alignment
        translate([0, 0, -0.1])
            cylinder(h = 1.2, r1 = ring_inner_radius + thread_depth + 0.5, r2 = ring_inner_radius);
    }
}

// ================================================
// HELICAL SQUARE THREAD GENERATOR
// ================================================

module square_thread(r, h, pitch, depth, internal = false, clearance = 0) {
    effective_r = internal ? (r + clearance) : r;
    turns = h / pitch;
    slices = floor(turns * 36); // Smooth helical resolution
    
    // Each tooth profile has a flat top and bottom with equal height (pitch/2)
    polyhedron(
        points = [
            for (i = [0 : slices]) 
                let (
                    a = i * (360 / 36),
                    z = i * (h / slices),
                    
                    // Taper profile over top and bottom turn to eliminate sharp overhangs
                    taper = (z < pitch) ? (z / pitch) : ((z > h - pitch) ? ((h - z) / pitch) : 1),
                    current_depth = depth * taper,
                    
                    r_inner = internal ? effective_r : effective_r - current_depth,
                    r_outer = internal ? effective_r + current_depth : effective_r
                )
                each [
                    // Bottom inner & outer vertices
                    [ r_inner * cos(a), r_inner * sin(a), z ],
                    [ r_outer * cos(a), r_outer * sin(a), z ],
                    
                    // Top inner & outer vertices (flat crest width = pitch / 2)
                    [ r_inner * cos(a), r_inner * sin(a), z + pitch/2 ],
                    [ r_outer * cos(a), r_outer * sin(a), z + pitch/2 ]
                ]
        ],
        faces = [
            for (i = [0 : slices - 1])
                let (b = i * 4)
                each [
                    // Quadrilateral face mesh construction
                    [b,   b+1, b+5, b+4],
                    [b+1, b+3, b+7, b+5],
                    [b+3, b+2, b+6, b+7],
                    [b+2, b,   b+4, b+6]
                ]
        ]
    );
}
